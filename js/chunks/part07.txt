function comboPartner(u,target){
 return living().find(x=>x.side==='player'&&!x.isEscort&&x.id!==u.id&&dist(x,u)===1&&dist(x,target)<=attackRange(x)&&!x.acted);
}
function tryCombo(u,target){
 const p=comboPartner(u,target);if(!p||Math.random()>.45)return false;
 const dmg1=calcDamage(u,target),dmg2=Math.max(1,Math.round(calcDamage(p,target)*.8));
 target.hp=Math.max(0,target.hp-dmg1-dmg2);p.acted=true;
 document.querySelector('.boardWrap').classList.add('comboFlash');setTimeout(()=>document.querySelector('.boardWrap').classList.remove('comboFlash'),800);
 showCutIn(u,`${p.name}と連携！`);combatFx('連携攻撃！');
 log(`🤝 ${u.name}と${p.name}の連携攻撃！ ${target.name}に合計${dmg1+dmg2}ダメージ！`,'info');
 updateMorale(3);return true;
}
function specialName(u){
 if(u.id==='liu')return '仁徳の号令';
 if(u.id==='guan')return '青龍偃月斬';
 if(u.id==='zhang')return '蛇矛猛撃';
 if(u.id==='zhao')return '龍胆突撃';
 if(u.id==='tian')return '連弩射撃';
 return '奮戦';
}
function useSpecial(){
 if(!selected||selected.acted||selected.mp<6)return;
 const u=selected;u.mp-=6;
 document.querySelector('.boardWrap').classList.add('specialFx');
 setTimeout(()=>document.querySelector('.boardWrap').classList.remove('specialFx'),800);
 showCutIn(u,specialName(u)+'！');
 if(u.id==='liu'){
  living().filter(x=>x.side==='player'&&dist(x,u)<=2).forEach(x=>x.hp=Math.min(x.maxHp,x.hp+10));
  combatFx('仁徳の号令！');log('🌟 劉備の仁徳の号令！ 周囲の味方を回復した。');
 }else if(u.id==='tian'){
  const ts=living().filter(x=>x.side==='enemy'&&dist(x,u)<=2).slice(0,3);
  ts.forEach(t=>{const dmg=Math.max(5,Math.round(atk(u)*.85)-Math.floor(def(t)*.35));t.hp=Math.max(0,t.hp-dmg);log(`🏹 ${t.name}に${dmg}ダメージ。`)});
  combatFx('連弩射撃！');
 }else{
  const ts=living().filter(x=>x.side==='enemy'&&dist(x,u)<=attackRange(u));
  if(!ts.length){u.mp+=6;log('必殺技の対象がいない。');render();return}
  const t=ts.sort((a,b)=>a.hp-b.hp)[0];
  const mult=u.id==='guan'?2.0:u.id==='zhang'?1.9:u.id==='zhao'?1.8:1.6;
  const dmg=Math.max(8,Math.round((atk(u)+4-def(t)*.4)*mult));t.hp=Math.max(0,t.hp-dmg);
  combatFx(specialName(u)+'！');log(`💥 ${u.name}の${specialName(u)}！ ${t.name}に${dmg}ダメージ！`);
  if(t.hp<=0){gainExp(u,65);log(`⚔ ${t.name}を撃破！`)}
 }
 u.acted=true;checkResult();postPlayerAction();
}
function triggerEvent(){

 const k=key(selected.x,selected.y),ev=stages[stageIndex].events[k];if(!ev||visitedEvents.has(k))return;
 visitedEvents.add(k);log((ev.kind==='chest'?'🧰 ':'🏠 ')+ev.text.replace(/「|」/g,''));
 if(ev.item){
  if(['鉄の剣','鋼の剣','革の鎧','鉄の鎧'].includes(ev.item)){equipFound(selected,ev.item)}
  else {selected.items.push(ev.item);log(`🎁 ${ev.item}を入手した。`)}
 }
 if(ev.mp){selected.mp=Math.min(selected.maxMp,selected.mp+ev.mp);log(`🔷 ${selected.name}の気力が${ev.mp}回復した。`)}
 if(ev.recruit==='zhao'&&!getUnit('zhao')){
  const z=zhaoUnit();z.x=Math.max(0,selected.x-1);z.y=selected.y;units.push(z);log('⭐ 趙雲が仲間になった！');
 }
 if(ev.recruit==='zhuge'&&!getUnit('zhuge')){
  const z=zhugeUnit();z.x=Math.max(0,selected.x-1);z.y=selected.y;units.push(z);log('🪶 諸葛亮が仲間になった！ 軍師専用策略が使用可能になった。','info');

 }
 selected.acted=true;postPlayerAction();
}
