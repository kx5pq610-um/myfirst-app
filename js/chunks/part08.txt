function calcDamage(a,d){
 const variance=Math.floor(Math.random()*3)-1+(currentTime==='night'&&a.side==='enemy'?(Math.random()<.35?-2:0):0);
 let terrainBonus=terrain[d.y][d.x]==='fort'?2:terrain[d.y][d.x]==='forest'?1:terrain[d.y][d.x]==='pass'?2:0;
 const base=Math.max(1,atk(a)-(def(d)+terrainBonus)+3+variance);
 return Math.max(1,Math.round(base*troopMultiplier(a,d)*(a.side==='player'?moraleMod():enemyMoraleMod())))
}
function battle(a,d){
 const mul=troopMultiplier(a,d);
 const critChance=a.id==='j6'?0.24:(a.id==='guan'||a.id==='zhang'||a.id==='zhao'?0.16:0.10);
 const critical=Math.random()<critChance;
 const label=critical?'会心！':mul>1?'有利！':mul<1?'不利':'攻撃！';
 combatFx(label);
 if(critical)showCutIn(a,'会心の一撃！');
 let dmg=calcDamage(a,d);if(critical)dmg=Math.round(dmg*1.7);
 d.hp=Math.max(0,d.hp-dmg);log(`${critical?'💥 ':''}${a.name}（${a.troop||'歩兵'}）の攻撃！ ${d.name}に${dmg}のダメージ。`,'hit');
 a.tempAtk=0;
 if(d.hp<=0){
   log(`⚔ ${d.name}を撃破！`);
   if(a.side==='player'){gainExp(a,45);updateMorale(5);enemyMorale=Math.max(0,enemyMorale-6);battleStats.kills++}
   else updateMorale(-8);
   if(!bossBonusClaimed&&d.role&&d.role.includes('敵将')){bossBonusClaimed=true;money+=300;updateMorale(10);log('🏆 敵将撃破ボーナス！ 軍資金+300、士気+10','info')}
   checkResult();return
 }
 const range=dist(a,d);
 if(range<=attackRange(d)){
  const counter=calcDamage(d,a);a.hp=Math.max(0,a.hp-counter);if(a.side==='player')battleStats.damageTaken+=counter;log(`${d.name}の反撃！ ${a.name}に${counter}のダメージ。`);
  if(a.hp<=0){log(`☠ ${a.name}が倒れた。`);checkResult()}else if(a.side==='player')gainExp(a,15);
 }else{
  log(`${d.name}は射程外のため反撃できない。`);
 }
}

function tileKey(x,y){return `${x},${y}`}


function igniteTile(x,y,turns=3){
 if(!inBounds(x,y))return;
 if(['water','mountain'].includes(terrain[y][x]))return;
 burningTiles[tileKey(x,y)]=Math.max(burningTiles[tileKey(x,y)]||0,turns);
}
function spreadFire(){
 if(currentWeather!=='wind')return;
 const additions=[];
 Object.keys(burningTiles).forEach(k=>{
  const [x,y]=k.split(',').map(Number);
  [[1,0],[-1,0],[0,1],[0,-1]].forEach(([dx,dy])=>{
   const nx=x+dx,ny=y+dy;if(!inBounds(nx,ny))return;
   if(['ship','forest'].includes(terrain[ny][nx])&&Math.random()<.45)additions.push([nx,ny]);
  });
 });
 additions.forEach(([x,y])=>igniteTile(x,y,2));
}
function applyBurningTiles(){
 Object.keys(burningTiles).forEach(k=>{
  burningTiles[k]--;
  const [x,y]=k.split(',').map(Number),u=unitAt(x,y);
  if(u&&u.hp>0){
   const dmg=Math.max(4,Math.round(u.maxHp*.1));u.hp=Math.max(0,u.hp-dmg);
   log(`🔥 ${u.name}は燃える地形で${dmg}ダメージ。`,'bad');
  }
  if(burningTiles[k]<=0)delete burningTiles[k];
 });
 spreadFire();
}

}

}
