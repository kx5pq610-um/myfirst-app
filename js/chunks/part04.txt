  const d=Math.max(2,Math.round(u.maxHp*.08));u.hp=Math.max(0,u.hp-d);u.burnTurns--;
  log(`🔥 ${u.name}は炎上で${d}のダメージ。`,'bad');
 }
 if(u.poisonTurns>0){
  const d=Math.max(1,Math.round(u.maxHp*.06));u.hp=Math.max(0,u.hp-d);u.poisonTurns--;
  log(`☠ ${u.name}は毒で${d}のダメージ。`,'bad');
 }
