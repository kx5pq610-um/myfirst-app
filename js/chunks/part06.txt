function living(){return units.filter(u=>u.hp>0)}
function unitAt(x,y){return living().find(u=>u.x===x&&u.y===y)}
function getUnit(id){return units.find(u=>u.id===id)}
function key(x,y){return `${x},${y}`}
function inBounds(x,y){return x>=0&&y>=0&&x<COLS&&y<ROWS}
function dist(a,b){return Math.abs(a.x-b.x)+Math.abs(a.y-b.y)}

function reachable(u){
 const out=new Map([[key(u.x,u.y),0]]),q=[[u.x,u.y,0]];
 while(q.length){
  const [x,y,c]=q.shift();
  for(const [dx,dy] of [[1,0],[-1,0],[0,1],[0,-1]]){
   const nx=x+dx,ny=y+dy;if(!inBounds(nx,ny))continue;
   let tc=terrainCost[terrain[ny][nx]];
   if(stages[stageIndex].naval&&terrain[ny][nx]==='water')tc=2;
   if(tc>=99)continue;
   const occ=unitAt(nx,ny);if(occ&&occ.id!==u.id)continue;
   let stepCost=terrainCost[terrain[ny][nx]];
   if(stages[stageIndex].naval&&terrain[ny][nx]==='water')stepCost=2;
   const nc=c+stepCost;if(nc>(u.move+(u.tempMove||0)))continue;
   const k=key(nx,ny);if(!out.has(k)||nc<out.get(k)){out.set(k,nc);q.push([nx,ny,nc])}
  }
 }
 return out;
}
